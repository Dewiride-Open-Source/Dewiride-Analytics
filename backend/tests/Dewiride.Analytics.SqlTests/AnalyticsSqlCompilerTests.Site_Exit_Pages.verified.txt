--- statement ---
WITH
    windowed AS
    (
        SELECT
            event_id,
            surface,
            visitor_key,
            correlation_id,
            server_ts,
            kind,
            path
        FROM events
        WHERE site_id = {site_id:UUID}
          AND server_ts >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
          AND server_ts < fromUnixTimestamp64Milli({to_ms:Int64} + {idle_seconds:Int64} * 1000, 'UTC')
    ),
    stamped AS
    (
        SELECT
            correlation_id,
            visitor_key,
            surface IN ('BrowserTracker', 'NoScriptPixel') AS from_browser,
            toUnixTimestamp64Milli(server_ts) AS at_ms
        FROM windowed
        WHERE correlation_id != ''
          AND visitor_key != ''
    ),
    bundled AS
    (
        SELECT
            correlation_id AS name,
            groupArray((visitor_key, at_ms, from_browser)) AS reports
        FROM stamped
        GROUP BY correlation_id
        HAVING countIf(NOT from_browser) > 0
    ),
    echoed AS
    (
        SELECT
            name,
            delivered.1 AS reported_key,
            delivered.2 AS delivery_ms,
            arrayFilter(report -> abs(report.2 - delivery_ms) <= 120000, reports) AS around,
            arrayDistinct(arrayMap(report -> report.1, arrayFilter(report -> NOT report.3, around))) AS reporters,
            arrayDistinct(arrayMap(report -> report.1, arrayFilter(report -> report.3, around))) AS answerers,
            if(length(reporters) = 1 AND length(answerers) = 1, answerers[1], '') AS answered_by
        FROM bundled
        ARRAY JOIN arrayDistinct(arrayMap(report -> (report.1, report.2), arrayFilter(report -> NOT report.3, reports))) AS delivered
    ),
    timeline AS
    (
        SELECT
            visitor_key AS reporter,
            toUnixTimestamp64Milli(server_ts) AS at_ms,
            '' AS identifier,
            toUInt8(0) AS is_name,
            '' AS answerer
        FROM windowed
        WHERE surface NOT IN ('BrowserTracker', 'NoScriptPixel')
          AND correlation_id = ''
          AND (visitor_key) IN
          (
              SELECT visitor_key
              FROM stamped
              WHERE NOT from_browser
          )
        GROUP BY reporter, at_ms
        UNION ALL
        SELECT
            reported_key AS reporter,
            delivery_ms AS at_ms,
            name AS identifier,
            toUInt8(1) AS is_name,
            answered_by AS answerer
        FROM echoed
    ),
    tallied AS
    (
        SELECT
            *,
            -at_ms AS backwards_ms,
            if(answerer = '', 0, row_number() OVER (
                PARTITION BY reporter, answerer
                ORDER BY at_ms, identifier)) AS answer_number
        FROM timeline
    ),
    weighed AS
    (
        SELECT
            reporter,
            at_ms,
            identifier,
            sum(is_name) OVER ahead - sum(is_name) OVER behind AS names_near,
            countIf(answerer != '') OVER ahead - countIf(answerer != '') OVER behind AS answered_near,
            argMaxIf((answerer, answer_number), (at_ms, identifier), answerer != '') OVER ahead AS last_answer,
            argMinIf((answerer, answer_number), (at_ms, identifier), answerer != '') OVER since AS first_answer,
            if(answered_near > 0
                   AND first_answer.1 = last_answer.1
                   AND last_answer.2 - first_answer.2 + 1 = answered_near
                   AND 2 * answered_near >= names_near
                   AND last_answer.1 != reporter,
               last_answer.1,
               '') AS browser_key
        FROM tallied
        WINDOW
            ahead AS (PARTITION BY reporter ORDER BY at_ms RANGE BETWEEN UNBOUNDED PRECEDING AND 1680000 FOLLOWING),
            behind AS (PARTITION BY reporter ORDER BY at_ms RANGE BETWEEN UNBOUNDED PRECEDING AND 1680001 PRECEDING),
            since AS (PARTITION BY reporter ORDER BY backwards_ms RANGE BETWEEN UNBOUNDED PRECEDING AND 1680000 FOLLOWING)
    ),
    linked AS
    (
        SELECT
            reporter AS reported_key,
            at_ms AS changed_ms,
            browser_key
        FROM
        (
            SELECT
                reporter,
                at_ms,
                browser_key,
                lagInFrame(browser_key, 1, '') OVER (
                    PARTITION BY reporter
                    ORDER BY at_ms, identifier
                    ROWS BETWEEN 1 PRECEDING AND CURRENT ROW) AS before
            FROM weighed
        )
        WHERE browser_key != before
    ),
    identified AS
    (
        SELECT
            windowed.* EXCEPT (visitor_key),
            if(linked.browser_key != '' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'), linked.browser_key, windowed.visitor_key) AS visitor_key
        FROM windowed
        ASOF LEFT JOIN linked ON windowed.visitor_key = linked.reported_key
            AND toUnixTimestamp64Milli(windowed.server_ts) >= linked.changed_ms
    ),
    reported AS
    (
        SELECT
            *,
            max(if(kind = 'PageView', toUnixTimestamp64Milli(server_ts), 0)) OVER (
                PARTITION BY visitor_key, path
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS arrival_ms
        FROM identified
        WHERE visitor_key != ''
    ),
    ordered AS
    (
        SELECT
            *,
            dateDiff('second', lagInFrame(server_ts, 1, server_ts) OVER visit, server_ts) AS since_previous
        FROM reported
        WHERE arrival_ms > 0
        WINDOW visit AS (PARTITION BY visitor_key ORDER BY server_ts, kind != 'PageView', event_id ROWS BETWEEN 1 PRECEDING AND CURRENT ROW)
    ),
    numbered AS
    (
        SELECT
            *,
            sum(toUInt8(kind = 'PageView' AND since_previous > {idle_seconds:Int64})) OVER (
                PARTITION BY visitor_key
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS arrival_ordinal
        FROM ordered
    ),
    grouped AS
    (
        SELECT
            *,
            max(if(kind = 'PageView', arrival_ordinal, 0)) OVER (
                PARTITION BY visitor_key, path
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS visit_ordinal
        FROM numbered
    ),
    sighted AS
    (
        SELECT
            *,
            row_number() OVER (
                PARTITION BY visitor_key, visit_ordinal, path, kind, surface IN ('BrowserTracker', 'NoScriptPixel')
                ORDER BY server_ts, event_id) AS sighting,
            sum(toUInt8(kind = 'PageView' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'))) OVER (
                PARTITION BY visitor_key, visit_ordinal, path) AS sightings_from_path
        FROM grouped
    ),
    counted AS
    (
        SELECT
            *,
            kind = 'PageView'
                AND surface IN ('BrowserTracker', 'NoScriptPixel')
                AND sighting <= sightings_from_path AS is_second_sighting
        FROM sighted
    ),
    paged AS
    (
        SELECT
            *,
            greatest(
                toUInt64(1),
                sum(toUInt8(kind = 'PageView' AND NOT is_second_sighting)) OVER (
                    PARTITION BY visitor_key, visit_ordinal, path
                    ORDER BY server_ts, event_id
                    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)) AS page_ordinal
        FROM counted
    ),
    opened AS
    (
        SELECT
            *,
            row_number() OVER (
                PARTITION BY visitor_key, visit_ordinal, path, page_ordinal
                ORDER BY is_second_sighting, kind != 'PageView', server_ts, event_id) = 1 AS opens_page
        FROM paged
    ),
    reconstructed AS
    (
        SELECT
            min(server_ts) AS started_at,
            max(server_ts) AS ended_at,
            toInt64(countIf(opens_page)) AS page_count,
            argMinIf(path, (server_ts, event_id), opens_page) AS entry_path,
            argMaxIf(path, (server_ts, event_id), opens_page) AS exit_path
        FROM opened
        GROUP BY visitor_key, visit_ordinal
        HAVING started_at >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
           AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
           AND ended_at < fromUnixTimestamp64Milli({settled_ms:Int64}, 'UTC')
    )
SELECT
    path,
    visits,
    toInt64(sum(visits) OVER ()) AS total_visits,
    toInt64(count() OVER ()) AS total_paths,
    toInt64(max(visits) OVER ()) AS most_visits
FROM
(
    SELECT
        exit_path AS path,
        toInt64(count()) AS visits
    FROM reconstructed
    GROUP BY path
)
ORDER BY visits DESC, path
LIMIT {limit:UInt32} OFFSET {offset:UInt32}
