--- statement ---
WITH
    windowed AS
    (
        SELECT kind, surface, path, visitor_key, correlation_id, server_ts
        FROM events
        WHERE site_id = {site_id:UUID}
          AND server_ts >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
          AND server_ts < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
    ),
    echoed AS
    (
        SELECT
            minIf(visitor_key, surface IN ('BrowserTracker', 'NoScriptPixel')) AS browser_key,
            minIf(visitor_key, surface NOT IN ('BrowserTracker', 'NoScriptPixel')) AS reported_key
        FROM windowed
        WHERE correlation_id != ''
          AND visitor_key != ''
        GROUP BY correlation_id
        HAVING browser_key != ''
           AND reported_key != ''
    ),
    linked AS
    (
        SELECT
            reported_key,
            min(browser_key) AS browser_key
        FROM echoed
        GROUP BY reported_key
    ),
    identified AS
    (
        SELECT
            windowed.* EXCEPT (visitor_key),
            if(linked.browser_key != '' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'), linked.browser_key, windowed.visitor_key) AS visitor_key
        FROM windowed
        LEFT JOIN linked ON windowed.visitor_key = linked.reported_key
    )
SELECT
    bucket,
    toInt64(uniqExactIf(visitor_key, visitor_key != '')) AS value
FROM
(
    SELECT
        toStartOfHour(server_ts, {time_zone:String}) AS bucket,
        visitor_key,
        greatest(
            countIf(kind = 'PageView' AND surface IN ('BrowserTracker', 'NoScriptPixel')),
            toUInt64(countIf(kind != 'PageView' AND surface IN ('BrowserTracker', 'NoScriptPixel')) > 0),
            countIf(kind = 'PageView' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'))) AS delivered,
        if(visitor_key = '', countIf(kind = 'PageView'), delivered) AS page_views
    FROM identified
    GROUP BY bucket, visitor_key, path
)
GROUP BY bucket
ORDER BY bucket
WITH FILL
    FROM toStartOfHour(fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC'), {time_zone:String})
    TO toStartOfHour(fromUnixTimestamp64Milli({to_ms:Int64} - 1, 'UTC'), {time_zone:String}) + INTERVAL 1 HOUR
    STEP INTERVAL 1 HOUR

--- bound values ---
site_id = 0197c0de-0000-7000-8000-000000000001
from_ms = 1777593600000
to_ms = 1778198400000
time_zone = 'Europe/London'
