WITH
    windowed AS
    (
        SELECT kind, surface, path, visitor_key, correlation_id, server_ts
        FROM events
        WHERE site_id = {site_id:UUID}
          AND server_ts >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
          AND server_ts < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
    ),
    echoed AS
    (
        SELECT
            minIf(visitor_key, surface IN ('BrowserTracker', 'NoScriptPixel')) AS browser_key,
            minIf(visitor_key, surface NOT IN ('BrowserTracker', 'NoScriptPixel')) AS reported_key
        FROM windowed
        WHERE correlation_id != ''
          AND visitor_key != ''
        GROUP BY correlation_id
        HAVING browser_key != ''
           AND reported_key != ''
    ),
    linked AS
    (
        SELECT
            reported_key,
            min(browser_key) AS browser_key
        FROM echoed
        GROUP BY reported_key
    ),
    identified AS
    (
        SELECT
            windowed.* EXCEPT (visitor_key),
            if(linked.browser_key != '' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'), linked.browser_key, windowed.visitor_key) AS visitor_key
        FROM windowed
        LEFT JOIN linked ON windowed.visitor_key = linked.reported_key
    ),
    people AS
    (
        SELECT
            splitByChar(':', session_key)[1] AS visitor_key,
            began,
            ended
        FROM
        (
            SELECT
                session_key,
                argMax(category, (ruleset_major, ruleset_minor, classified_at)) AS category,
                argMax(started_at, (ruleset_major, ruleset_minor, classified_at)) AS began,
                argMax(ended_at, (ruleset_major, ruleset_minor, classified_at)) AS ended
            FROM session_classifications
            WHERE site_id = {site_id:UUID}
              AND started_at >= fromUnixTimestamp64Milli({from_ms:Int64} - {longest_visit_seconds:Int64} * 1000, 'UTC')
              AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
            GROUP BY session_key
        )
        WHERE category = 'LikelyHuman'
    ),
    attributed AS
    (
        SELECT identified.*
        FROM identified
        INNER JOIN people ON identified.visitor_key = people.visitor_key
        WHERE identified.server_ts >= people.began
          AND identified.server_ts <= people.ended
    )
SELECT
    cut,
    day,
    slot,
    page_views,
    visitors
FROM
(
    SELECT
        'month' AS cut,
        toDate(0) AS day,
        toUInt8(0) AS slot,
        toInt64(sum(page_views)) AS page_views,
        toInt64(uniqExactIf(visitor_key, visitor_key != '')) AS visitors
    FROM
    (
        SELECT
            visitor_key,
            greatest(
                countIf(kind = 'PageView' AND surface IN ('BrowserTracker', 'NoScriptPixel')),
                countIf(kind = 'PageView' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'))) AS delivered,
            if(visitor_key = '', countIf(kind = 'PageView'), delivered) AS page_views
        FROM attributed
        GROUP BY visitor_key, path
    )
    UNION ALL
    SELECT
        'day' AS cut,
        calendar_day AS day,
        toUInt8(toDayOfWeek(calendar_day)) AS slot,
        toInt64(sum(page_views)) AS page_views,
        toInt64(0) AS visitors
    FROM
    (
        SELECT
            toDate(server_ts, {time_zone:String}) AS calendar_day,
            visitor_key,
            greatest(
                countIf(kind = 'PageView' AND surface IN ('BrowserTracker', 'NoScriptPixel')),
                countIf(kind = 'PageView' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'))) AS delivered,
            if(visitor_key = '', countIf(kind = 'PageView'), delivered) AS page_views
        FROM attributed
        GROUP BY calendar_day, visitor_key, path
    )
    GROUP BY calendar_day
    UNION ALL
    SELECT
        'hour' AS cut,
        toDate(calendar_hour, {time_zone:String}) AS day,
        toUInt8(toHour(calendar_hour, {time_zone:String})) AS slot,
        toInt64(sum(page_views)) AS page_views,
        toInt64(0) AS visitors
    FROM
    (
        SELECT
            toStartOfHour(server_ts, {time_zone:String}) AS calendar_hour,
            visitor_key,
            greatest(
                countIf(kind = 'PageView' AND surface IN ('BrowserTracker', 'NoScriptPixel')),
                countIf(kind = 'PageView' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'))) AS delivered,
            if(visitor_key = '', countIf(kind = 'PageView'), delivered) AS page_views
        FROM attributed
        GROUP BY calendar_hour, visitor_key, path
    )
    GROUP BY calendar_hour
)
SETTINGS max_threads = 1, max_execution_time = 600

time_zone = 'Asia/Kolkata'
longest_visit_seconds = 86400
