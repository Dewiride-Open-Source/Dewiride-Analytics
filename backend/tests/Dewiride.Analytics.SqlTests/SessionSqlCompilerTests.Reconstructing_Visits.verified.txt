WITH
    arrived AS
    (
        SELECT
            event_id,
            visitor_key,
            correlation_id,
            server_ts,
            kind,
            path,
            status_code,
            surface,
            user_agent,
            language,
            viewport_width,
            engaged_ms,
            scroll_depth_percent,
            had_pointer_interaction,
            had_keyboard_interaction,
            declared_web_driver,
            autonomous_system,
            network_owner,
            confirmed_operator,
            ip_address,
            referrer AS source_address,
            if(
                referrer_domain != ''
                AND referrer_domain != {site_domain:String}
                AND NOT endsWith(referrer_domain, concat('.', {site_domain:String})),
                if(
                    startsWith(referrer_domain, 'www.'),
                    substring(referrer_domain, 5),
                    referrer_domain),
                '') AS sending_host
        FROM events
        WHERE site_id = {site_id:UUID}
          AND server_ts >= fromUnixTimestamp64Milli({from_ms:Int64} - {longest_visit_seconds:Int64} * 1000, 'UTC')
          AND server_ts < fromUnixTimestamp64Milli({to_ms:Int64} + {longest_visit_seconds:Int64} * 1000, 'UTC')
    ),
    named AS
    (
        SELECT
            *,
            splitByChar('.', sending_host) AS labels,
            multiIf(
                length(labels) < 2, sending_host,
                length(labels) > 2
                    AND has({second_levels:Array(String)}, arrayElement(labels, -2)),
                    arrayElement(labels, -3),
                arrayElement(labels, -2)) AS sending_name,
            if(
                has({source_keys:Array(String)}, sending_host),
                sending_host,
                sending_name) AS catalogue_key
        FROM arrived
    ),
    windowed AS
    (
        SELECT
            event_id,
            visitor_key,
            correlation_id,
            server_ts,
            kind,
            path,
            status_code,
            surface,
            user_agent,
            language,
            viewport_width,
            engaged_ms,
            scroll_depth_percent,
            had_pointer_interaction,
            had_keyboard_interaction,
            declared_web_driver,
            autonomous_system,
            network_owner,
            confirmed_operator,
            ip_address,
            source_address,
            sending_host,
            transform(
                catalogue_key,
                {source_keys:Array(String)},
                {source_names:Array(String)},
                sending_host) AS source_site,
            if(
                sending_host = '',
                '',
                transform(
                    catalogue_key,
                    {source_keys:Array(String)},
                    {source_channels:Array(String)},
                    'link')) AS source_channel
        FROM named
    ),
    echoed AS
    (
        SELECT
            minIf(visitor_key, surface IN ('BrowserTracker', 'NoScriptPixel')) AS browser_key,
            minIf(visitor_key, surface NOT IN ('BrowserTracker', 'NoScriptPixel')) AS reported_key
        FROM windowed
        WHERE correlation_id != ''
          AND visitor_key != ''
        GROUP BY correlation_id
        HAVING browser_key != ''
           AND reported_key != ''
    ),
    linked AS
    (
        SELECT
            reported_key,
            min(browser_key) AS browser_key
        FROM echoed
        GROUP BY reported_key
    ),
    identified AS
    (
        SELECT
            windowed.* EXCEPT (visitor_key),
            if(linked.browser_key != '' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'), linked.browser_key, windowed.visitor_key) AS visitor_key
        FROM windowed
        LEFT JOIN linked ON windowed.visitor_key = linked.reported_key
    ),
    reported AS
    (
        SELECT
            *,
            max(if(kind = 'PageView', toUnixTimestamp64Milli(server_ts), 0)) OVER (
                PARTITION BY visitor_key, path
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS arrival_ms
        FROM identified
        WHERE visitor_key != ''
    ),
    ordered AS
    (
        SELECT
            *,
            dateDiff('second', lagInFrame(server_ts, 1, server_ts) OVER visit, server_ts) AS since_previous
        FROM reported
        WHERE arrival_ms > 0
        WINDOW visit AS (PARTITION BY visitor_key ORDER BY server_ts, kind != 'PageView', event_id ROWS BETWEEN 1 PRECEDING AND CURRENT ROW)
    ),
    numbered AS
    (
        SELECT
            *,
            sum(toUInt8(kind = 'PageView' AND since_previous > {idle_seconds:Int64})) OVER (
                PARTITION BY visitor_key
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS arrival_ordinal
        FROM ordered
    ),
    grouped AS
    (
        SELECT
            *,
            max(if(kind = 'PageView', arrival_ordinal, 0)) OVER (
                PARTITION BY visitor_key, path
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS visit_ordinal
        FROM numbered
    ),
    sighted AS
    (
        SELECT
            *,
            row_number() OVER (
                PARTITION BY visitor_key, visit_ordinal, path, kind, surface IN ('BrowserTracker', 'NoScriptPixel')
                ORDER BY server_ts, event_id) AS sighting,
            sum(toUInt8(kind = 'PageView' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'))) OVER (
                PARTITION BY visitor_key, visit_ordinal, path) AS sightings_from_path
        FROM grouped
    ),
    counted AS
    (
        SELECT
            *,
            kind = 'PageView'
                AND surface IN ('BrowserTracker', 'NoScriptPixel')
                AND sighting <= sightings_from_path AS is_second_sighting
        FROM sighted
    ),
    paged AS
    (
        SELECT
            *,
            greatest(
                toUInt64(1),
                sum(toUInt8(kind = 'PageView' AND NOT is_second_sighting)) OVER (
                    PARTITION BY visitor_key, visit_ordinal, path
                    ORDER BY server_ts, event_id
                    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)) AS page_ordinal
        FROM counted
    ),
    opened AS
    (
        SELECT
            *,
            row_number() OVER (
                PARTITION BY visitor_key, visit_ordinal, path, page_ordinal
                ORDER BY is_second_sighting, kind != 'PageView', server_ts, event_id) = 1 AS opens_page
        FROM paged
    ),
    attended AS
    (
        SELECT
            *,
            max(engaged_ms) OVER (
                PARTITION BY visitor_key, visit_ordinal, path, page_ordinal) AS page_engaged_ms
        FROM opened
    ),
    visits AS
    (
        SELECT
            concat(visitor_key, ':', toString(toUnixTimestamp64Milli(min(server_ts)))) AS session_key,
            min(server_ts) AS started_at,
            max(server_ts) AS ended_at,
            toBool(max(server_ts) < fromUnixTimestamp64Milli({settled_ms:Int64}, 'UTC')) AS is_closed,
            toUInt32(countIf(opens_page)) AS page_count,
            groupArraySortedIf({max_requests:UInt32})(
                (toUnixTimestamp64Milli(server_ts), path, status_code),
                opens_page) AS requests,
            groupUniqArray(toString(surface)) AS surfaces,
            anyIf(user_agent, user_agent != '') AS user_agent,
            anyIf(language, language != '') AS language,
            max(viewport_width) AS viewport_width,
            sumIf(page_engaged_ms, opens_page) AS engaged_ms,
            max(scroll_depth_percent) AS max_scroll_depth_percent,
            toUInt32(countIf(had_pointer_interaction != 'Unobserved')) AS pointer_observed,
            toUInt32(countIf(had_pointer_interaction = 'Yes')) AS pointer_seen,
            toUInt32(countIf(had_keyboard_interaction != 'Unobserved')) AS keyboard_observed,
            toUInt32(countIf(had_keyboard_interaction = 'Yes')) AS keyboard_seen,
            toUInt32(countIf(declared_web_driver != 'Unobserved')) AS web_driver_observed,
            toUInt32(countIf(declared_web_driver = 'Yes')) AS web_driver_seen,
            max(autonomous_system) AS autonomous_system,
            anyIf(network_owner, network_owner != '') AS network_owner,
            anyIf(confirmed_operator, confirmed_operator != '') AS stamped_operator,
            argMinIf(source_channel, (server_ts, event_id), sending_host != '') AS from_kind,
            anyIf(ip_address, ip_address != '') AS ip_address
        FROM attended
        GROUP BY visitor_key, visit_ordinal
        HAVING started_at >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
           AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
           AND (session_key, ended_at) NOT IN
           (
               SELECT session_key, ended_at
               FROM session_classifications
               WHERE site_id = {site_id:UUID}
                 AND ruleset_major = {ruleset_major:UInt16}
                 AND ruleset_minor = {ruleset_minor:UInt16}
                 AND started_at >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
                 AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
           )
    ),
    confirmations AS
    (
        SELECT
            session_key,
            argMax(operator, settled_at) AS operator
        FROM name_confirmations
        WHERE site_id = {site_id:UUID}
          AND started_at >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
          AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
        GROUP BY session_key
    )
SELECT
    session_key,
    started_at,
    ended_at,
    is_closed,
    page_count,
    requests,
    surfaces,
    user_agent,
    language,
    viewport_width,
    engaged_ms,
    max_scroll_depth_percent,
    pointer_observed,
    pointer_seen,
    keyboard_observed,
    keyboard_seen,
    web_driver_observed,
    web_driver_seen,
    autonomous_system,
    network_owner,
    if(stamped_operator != '', stamped_operator, ifNull(operator, '')) AS confirmed_operator,
    from_kind,
    ip_address
FROM visits
LEFT JOIN confirmations USING (session_key)
ORDER BY started_at, session_key
