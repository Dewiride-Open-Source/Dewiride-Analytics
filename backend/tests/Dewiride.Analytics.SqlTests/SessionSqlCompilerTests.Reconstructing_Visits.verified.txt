WITH
    arrived AS
    (
        SELECT
            event_id,
            visitor_key,
            correlation_id,
            server_ts,
            client_ts,
            kind,
            path,
            status_code,
            surface,
            user_agent,
            language,
            viewport_width,
            viewport_height,
            engaged_ms,
            scroll_depth_percent,
            had_pointer_interaction,
            had_keyboard_interaction,
            declared_web_driver,
            autonomous_system,
            network_owner,
            confirmed_operator,
            ip_address,
            timezone_offset_minutes,
            browser_family,
            operating_system,
            referrer AS source_address,
            if(
                referrer_domain != ''
                AND referrer_domain != {site_domain:String}
                AND NOT endsWith(referrer_domain, concat('.', {site_domain:String})),
                if(
                    startsWith(referrer_domain, 'www.'),
                    substring(referrer_domain, 5),
                    referrer_domain),
                '') AS sending_host
        FROM events
        WHERE site_id = {site_id:UUID}
          AND server_ts >= fromUnixTimestamp64Milli({from_ms:Int64} - {longest_visit_seconds:Int64} * 1000, 'UTC')
          AND server_ts < fromUnixTimestamp64Milli({to_ms:Int64} + {longest_visit_seconds:Int64} * 1000, 'UTC')
    ),
    named AS
    (
        SELECT
            *,
            splitByChar('.', sending_host) AS labels,
            multiIf(
                length(labels) < 2, sending_host,
                length(labels) > 2
                    AND has({second_levels:Array(String)}, arrayElement(labels, -2)),
                    arrayElement(labels, -3),
                arrayElement(labels, -2)) AS sending_name,
            if(
                has({source_keys:Array(String)}, sending_host),
                sending_host,
                sending_name) AS catalogue_key
        FROM arrived
    ),
    windowed AS
    (
        SELECT
            event_id,
            visitor_key,
            correlation_id,
            server_ts,
            client_ts,
            kind,
            path,
            status_code,
            surface,
            user_agent,
            language,
            viewport_width,
            viewport_height,
            engaged_ms,
            scroll_depth_percent,
            had_pointer_interaction,
            had_keyboard_interaction,
            declared_web_driver,
            autonomous_system,
            network_owner,
            confirmed_operator,
            ip_address,
            timezone_offset_minutes,
            browser_family,
            operating_system,
            source_address,
            sending_host,
            transform(
                catalogue_key,
                {source_keys:Array(String)},
                {source_names:Array(String)},
                sending_host) AS source_site,
            if(
                sending_host = '',
                '',
                transform(
                    catalogue_key,
                    {source_keys:Array(String)},
                    {source_channels:Array(String)},
                    'link')) AS source_channel
        FROM named
    ),
    stamped AS
    (
        SELECT
            correlation_id,
            visitor_key,
            surface IN ('BrowserTracker', 'NoScriptPixel') AS from_browser,
            toUnixTimestamp64Milli(server_ts) AS at_ms
        FROM windowed
        WHERE correlation_id != ''
          AND visitor_key != ''
    ),
    bundled AS
    (
        SELECT
            correlation_id AS name,
            groupArray((visitor_key, at_ms, from_browser)) AS reports
        FROM stamped
        GROUP BY correlation_id
        HAVING countIf(NOT from_browser) > 0
    ),
    echoed AS
    (
        SELECT
            name,
            delivered.1 AS reported_key,
            delivered.2 AS delivery_ms,
            arrayFilter(report -> abs(report.2 - delivery_ms) <= 120000, reports) AS around,
            arrayDistinct(arrayMap(report -> report.1, arrayFilter(report -> NOT report.3, around))) AS reporters,
            arrayDistinct(arrayMap(report -> report.1, arrayFilter(report -> report.3, around))) AS answerers,
            if(length(reporters) = 1 AND length(answerers) = 1, answerers[1], '') AS answered_by
        FROM bundled
        ARRAY JOIN arrayDistinct(arrayMap(report -> (report.1, report.2), arrayFilter(report -> NOT report.3, reports))) AS delivered
    ),
    timeline AS
    (
        SELECT
            visitor_key AS reporter,
            toUnixTimestamp64Milli(server_ts) AS at_ms,
            '' AS identifier,
            toUInt8(0) AS is_name,
            '' AS answerer
        FROM windowed
        WHERE surface NOT IN ('BrowserTracker', 'NoScriptPixel')
          AND correlation_id = ''
          AND (visitor_key) IN
          (
              SELECT visitor_key
              FROM stamped
              WHERE NOT from_browser
          )
        GROUP BY reporter, at_ms
        UNION ALL
        SELECT
            reported_key AS reporter,
            delivery_ms AS at_ms,
            name AS identifier,
            toUInt8(1) AS is_name,
            answered_by AS answerer
        FROM echoed
    ),
    tallied AS
    (
        SELECT
            *,
            -at_ms AS backwards_ms,
            if(answerer = '', 0, row_number() OVER (
                PARTITION BY reporter, answerer
                ORDER BY at_ms, identifier)) AS answer_number
        FROM timeline
    ),
    weighed AS
    (
        SELECT
            reporter,
            at_ms,
            identifier,
            sum(is_name) OVER ahead - sum(is_name) OVER behind AS names_near,
            countIf(answerer != '') OVER ahead - countIf(answerer != '') OVER behind AS answered_near,
            argMaxIf((answerer, answer_number), (at_ms, identifier), answerer != '') OVER ahead AS last_answer,
            argMinIf((answerer, answer_number), (at_ms, identifier), answerer != '') OVER since AS first_answer,
            if(answered_near > 0
                   AND first_answer.1 = last_answer.1
                   AND last_answer.2 - first_answer.2 + 1 = answered_near
                   AND 2 * answered_near >= names_near
                   AND last_answer.1 != reporter,
               last_answer.1,
               '') AS browser_key
        FROM tallied
        WINDOW
            ahead AS (PARTITION BY reporter ORDER BY at_ms RANGE BETWEEN UNBOUNDED PRECEDING AND 1680000 FOLLOWING),
            behind AS (PARTITION BY reporter ORDER BY at_ms RANGE BETWEEN UNBOUNDED PRECEDING AND 1680001 PRECEDING),
            since AS (PARTITION BY reporter ORDER BY backwards_ms RANGE BETWEEN UNBOUNDED PRECEDING AND 1680000 FOLLOWING)
    ),
    linked AS
    (
        SELECT
            reporter AS reported_key,
            at_ms AS changed_ms,
            browser_key
        FROM
        (
            SELECT
                reporter,
                at_ms,
                browser_key,
                lagInFrame(browser_key, 1, '') OVER (
                    PARTITION BY reporter
                    ORDER BY at_ms, identifier
                    ROWS BETWEEN 1 PRECEDING AND CURRENT ROW) AS before
            FROM weighed
        )
        WHERE browser_key != before
    ),
    identified AS
    (
        SELECT
            windowed.* EXCEPT (visitor_key),
            if(linked.browser_key != '' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'), linked.browser_key, windowed.visitor_key) AS visitor_key
        FROM windowed
        ASOF LEFT JOIN linked ON windowed.visitor_key = linked.reported_key
            AND toUnixTimestamp64Milli(windowed.server_ts) >= linked.changed_ms
    ),
    reported AS
    (
        SELECT
            *,
            max(if(kind = 'PageView', toUnixTimestamp64Milli(server_ts), 0)) OVER (
                PARTITION BY visitor_key, path
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS arrival_ms
        FROM identified
        WHERE visitor_key != ''
    ),
    ordered AS
    (
        SELECT
            *,
            dateDiff('second', lagInFrame(server_ts, 1, server_ts) OVER visit, server_ts) AS since_previous
        FROM reported
        WHERE arrival_ms > 0
        WINDOW visit AS (PARTITION BY visitor_key ORDER BY server_ts, kind != 'PageView', event_id ROWS BETWEEN 1 PRECEDING AND CURRENT ROW)
    ),
    numbered AS
    (
        SELECT
            *,
            sum(toUInt8(kind = 'PageView' AND since_previous > {idle_seconds:Int64})) OVER (
                PARTITION BY visitor_key
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS arrival_ordinal
        FROM ordered
    ),
    grouped AS
    (
        SELECT
            *,
            max(if(kind = 'PageView', arrival_ordinal, 0)) OVER (
                PARTITION BY visitor_key, path
                ORDER BY server_ts, kind != 'PageView', event_id
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS visit_ordinal
        FROM numbered
    ),
    sighted AS
    (
        SELECT
            *,
            row_number() OVER (
                PARTITION BY visitor_key, visit_ordinal, path, kind, surface IN ('BrowserTracker', 'NoScriptPixel')
                ORDER BY server_ts, event_id) AS sighting,
            sum(toUInt8(kind = 'PageView' AND surface NOT IN ('BrowserTracker', 'NoScriptPixel'))) OVER (
                PARTITION BY visitor_key, visit_ordinal, path) AS sightings_from_path
        FROM grouped
    ),
    counted AS
    (
        SELECT
            *,
            kind = 'PageView'
                AND surface IN ('BrowserTracker', 'NoScriptPixel')
                AND sighting <= sightings_from_path AS is_second_sighting
        FROM sighted
    ),
    paged AS
    (
        SELECT
            *,
            greatest(
                toUInt64(1),
                sum(toUInt8(kind = 'PageView' AND NOT is_second_sighting)) OVER (
                    PARTITION BY visitor_key, visit_ordinal, path
                    ORDER BY server_ts, event_id
                    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)) AS page_ordinal
        FROM counted
    ),
    opened AS
    (
        SELECT
            *,
            row_number() OVER (
                PARTITION BY visitor_key, visit_ordinal, path, page_ordinal
                ORDER BY is_second_sighting, kind != 'PageView', server_ts, event_id) = 1 AS opens_page
        FROM paged
    ),
    attended AS
    (
        SELECT
            *,
            max(engaged_ms) OVER page AS page_engaged_ms,
            maxIf(engaged_ms, kind = 'Exit' AND surface = 'BrowserTracker') OVER page AS page_left_ms,
            max(server_ts) OVER page AS page_last_at,
            max(page_ordinal) OVER (PARTITION BY visitor_key, visit_ordinal, path) AS last_page_ordinal,
            minIf(client_ts, kind = 'PageView' AND surface = 'BrowserTracker') OVER page AS page_opened_on_client,
            argMinIf(
                (client_ts, engaged_ms),
                (server_ts, event_id),
                kind = 'Engagement' AND surface = 'BrowserTracker') OVER page AS first_progress
        FROM opened
        WINDOW page AS (PARTITION BY visitor_key, visit_ordinal, path, page_ordinal)
    ),
    visits AS
    (
        SELECT
            concat(visitor_key, ':', toString(toUnixTimestamp64Milli(min(server_ts)))) AS session_key,
            min(server_ts) AS started_at,
            max(server_ts) AS ended_at,
            toBool(max(server_ts) < fromUnixTimestamp64Milli({settled_ms:Int64}, 'UTC')) AS is_closed,
            toUInt32(countIf(opens_page)) AS page_count,
            groupArraySortedIf({max_requests:UInt32})(
                (toUnixTimestamp64Milli(server_ts), path, status_code),
                opens_page) AS requests,
            groupUniqArray(toString(surface)) AS surfaces,
            anyIf(user_agent, user_agent != '') AS user_agent,
            argMinIf(language, (server_ts, event_id), language != '') AS language,
            max(viewport_width) AS viewport_width,
            max(viewport_height) AS viewport_height,
            sumIf(page_engaged_ms, opens_page) AS engaged_ms,
            sumIf(
                if(
                    page_ordinal < last_page_ordinal
                    OR toDate(page_last_at + toIntervalSecond({progress_interval_seconds:Int64})) != toDate(page_last_at),
                    page_engaged_ms,
                    least(ifNull(page_left_ms, 0), page_engaged_ms)),
                opens_page) AS read_ms,
            minIf(
                greatest(
                    toUnixTimestamp64Milli(first_progress.1) - toUnixTimestamp64Milli(page_opened_on_client),
                    first_progress.2),
                opens_page
                AND page_ordinal = 1
                AND page_opened_on_client IS NOT NULL
                AND first_progress.1 IS NOT NULL
                AND first_progress.2 IS NOT NULL
                AND first_progress.1 >= page_opened_on_client) AS soonest_progress_ms,
            max(scroll_depth_percent) AS max_scroll_depth_percent,
            toUInt32(countIf(had_pointer_interaction != 'Unobserved')) AS pointer_observed,
            toUInt32(countIf(had_pointer_interaction = 'Yes')) AS pointer_seen,
            toUInt32(countIf(had_keyboard_interaction != 'Unobserved')) AS keyboard_observed,
            toUInt32(countIf(had_keyboard_interaction = 'Yes')) AS keyboard_seen,
            toUInt32(countIf(declared_web_driver != 'Unobserved')) AS web_driver_observed,
            toUInt32(countIf(declared_web_driver = 'Yes')) AS web_driver_seen,
            max(autonomous_system) AS autonomous_system,
            anyIf(network_owner, network_owner != '') AS network_owner,
            anyIf(confirmed_operator, confirmed_operator != '') AS stamped_operator,
            argMinIf(source_channel, (server_ts, event_id), sending_host != '') AS from_kind,
            anyIf(ip_address, ip_address != '') AS ip_address,
            if(
                min(timezone_offset_minutes) = max(timezone_offset_minutes),
                min(timezone_offset_minutes),
                NULL) AS clock_offset_minutes,
            argMinIf(browser_family, (server_ts, event_id), browser_family != '') AS browser,
            argMinIf(operating_system, (server_ts, event_id), operating_system != '') AS system_name
        FROM attended
        GROUP BY visitor_key, visit_ordinal
        HAVING started_at >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
           AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
           AND (session_key, ended_at) NOT IN
           (
               SELECT session_key, ended_at
               FROM session_classifications
               WHERE site_id = {site_id:UUID}
                 AND ruleset_major = {ruleset_major:UInt16}
                 AND ruleset_minor = {ruleset_minor:UInt16}
                 AND started_at >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
                 AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
           )
    ),
    confirmations AS
    (
        SELECT
            session_key,
            argMax(operator, settled_at) AS operator
        FROM name_confirmations
        WHERE site_id = {site_id:UUID}
          AND started_at >= fromUnixTimestamp64Milli({from_ms:Int64}, 'UTC')
          AND started_at < fromUnixTimestamp64Milli({to_ms:Int64}, 'UTC')
        GROUP BY session_key
    )
SELECT
    session_key,
    started_at,
    ended_at,
    is_closed,
    page_count,
    requests,
    surfaces,
    user_agent,
    language,
    viewport_width,
    engaged_ms,
    max_scroll_depth_percent,
    pointer_observed,
    pointer_seen,
    keyboard_observed,
    keyboard_seen,
    web_driver_observed,
    web_driver_seen,
    autonomous_system,
    network_owner,
    if(stamped_operator != '', stamped_operator, ifNull(operator, '')) AS confirmed_operator,
    from_kind,
    ip_address,
    read_ms,
    soonest_progress_ms,
    viewport_height,
    clock_offset_minutes,
    browser,
    system_name
FROM visits
LEFT JOIN confirmations USING (session_key)
ORDER BY started_at, session_key
